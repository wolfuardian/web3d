// 統一處理 mouse + touch 的拖曳工具 function makeDraggable(el, callbacks) { const start = (e) => { e.preventDefault(); el.classList.add('dragging'); el.setPointerCapture(e.pointerId); callbacks.onStart?.(e); }; const move = (e) => { if (!el.classList.contains('dragging')) return; callbacks.onMove?.(e); }; const end = (e) => { el.classList.remove('dragging'); el.releasePointerCapture(e.pointerId); callbacks.onEnd?.(e); }; el.addEventListener('pointerdown', start); el.addEventListener('pointermove', move); el.addEventListener('pointerup', end); el.addEventListener('pointercancel', end); }
